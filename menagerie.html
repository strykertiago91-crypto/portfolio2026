Menagerie.java 
import java.util.ArrayList;

public class Menagerie {
    ArrayList<Animal> listeAnimaux;

    public Menagerie() {
        this.listeAnimaux = new ArrayList<Animal>();
    }

    public void ajouterUnAnimal(Animal unAnimal) {
        this.listeAnimaux.add(unAnimal);
    }

    public void retirerUnAnimal(Animal unAnimal) {
        this.listeAnimaux.remove(unAnimal);
    }

    public int rechercherAnimal(String unNom) {
        if (listeAnimaux.size() == 0) return -1;
        int compteur = 0;
        while (compteur < listeAnimaux.size() &&
               !listeAnimaux.get(compteur).getNom().equals(unNom)) {
            compteur++;
        }
        return (compteur < listeAnimaux.size()) ? compteur : -1;
    }
}

        btnSupprimer = new JButton("Supprimer un animal");
        // ... ajout des listeners et du panel
    }

    public void actionPerformed(ActionEvent e) {
        if      (e.getSource() == btnAjouter)    new AjoutAnimalFenetre();
        else if (e.getSource() == btnRechercher) new RechercheAnimalFenetre();

AjoutAnimalFenetre.java 
public class AjoutAnimalFenetre extends JFrame
                                  implements ActionListener {

    private JTextField jtfNom, jtfEspece;
    private JButton    btnValider;

    public void actionPerformed(ActionEvent e) {
        if (e.getSource() == btnValider) {
            String nom    = jtfNom.getText();
            String espece = jtfEspece.getText();
            Animal nouvelAnimal = new Animal(nom, espece);
            Programme.maMenagerie.ajouterUnAnimal(nouvelAnimal);
            JOptionPane.showMessageDialog(this,
                "Animal ajouté avec succès !");
            this.dispose();
        }
    }
}
